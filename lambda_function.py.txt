import json
import boto3

rekognition = boto3.client('rekognition')
sns = boto3.client('sns')

SNS_TOPIC_ARN = 'arn:aws:sns:ap-south-1:907791519749:ImageProcessingAlerts'

def lambda_handler(event, context):

    bucket = event['Records'][0]['s3']['bucket']['name']
    image = event['Records'][0]['s3']['object']['key']

    response = rekognition.detect_labels(
        Image={
            'S3Object': {
                'Bucket': bucket,
                'Name': image
            }
        },
        MaxLabels=10
    )

    labels = []

    for label in response['Labels']:
        labels.append(label['Name'])

    message = "Image Uploaded: {}\n\nDetected Labels:\n{}".format(
        image,
        ", ".join(labels)
    )

    sns.publish(
        TopicArn=SNS_TOPIC_ARN,
        Subject='Image Processing Result',
        Message=message
    )

    return {
        'statusCode': 200,
        'body': json.dumps(labels)
    }
